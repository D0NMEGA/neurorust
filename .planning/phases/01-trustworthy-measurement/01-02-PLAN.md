```sh
{ echo "=== apt-cache policy rtla ==="
```

```sh
cd "$HOME/neurorust-recon-$(date -u +%Y-%m-%d)"

# Plain histogram (-h): one column per thread, no summary column.

    --policy=fifo --priority=99 --interval=200 --distance=0 \
    --histogram=400 --histfile=probe-cyclictest-h-60s.hist \
    --json=probe-cyclictest-60s.json --duration=60

# histofall (-H): same run with the extra right-hand summary column.
sudo cyclictest --mainaffinity=0,1 --affinity=6-11 --threads=6 --mlockall \
    --policy=fifo --priority=99 --interval=200 --distance=0 \
    --histofall=400 --histfile=probe-cyclictest-histofall-60s.hist \
    --duration=60

# Real /proc/interrupts for the D-15 diff tests.

# Tuning snapshot in a stable, parseable text form for the D-06 precondition tests.
{ for c in /sys/devices/system/cpu/cpu[0-9]*; do
      n="${c##*/cpu}"
      echo "cpu${n}.governor=$(cat "$c/cpufreq/scaling_governor" 2>/dev/null || echo unavailable)"
  done
  echo "intel_pstate.no_turbo=$(cat /sys/devices/system/cpu/intel_pstate/no_turbo 2>/dev/null || echo unavailable)"
  for s in /sys/devices/system/cpu/cpu0/cpuidle/state*; do
      echo "cstate.$(cat "$s/name")=disable:$(cat "$s/disable")"
  done
  echo "kernel.realtime=$(cat /sys/kernel/realtime 2>/dev/null || echo unavailable)"
  echo "systemd.default_target=$(systemctl get-default)"
  echo "tracing.current_tracer=$(sudo cat /sys/kernel/tracing/current_tracer 2>/dev/null || echo unavailable)"
  echo "power.ac_online=$(cat /sys/class/power_supply/AC*/online 2>/dev/null | head -1 || echo unavailable)"
  echo "power.battery_status=$(cat /sys/class/power_supply/BAT*/status 2>/dev/null | head -1 || echo unavailable)"
  echo "power.battery_capacity=$(cat /sys/class/power_supply/BAT*/capacity 2>/dev/null | head -1 || echo unavailable)"
  for z in /sys/class/thermal/thermal_zone*; do
      echo "thermal.$(cat "$z/type")=$(cat "$z/temp")"
  done
} > probe-sysfs-tuning.txt
```

## Trust boundaries

| Boundary | Description |
|----------|-------------|
| rig -> public repository | Everything committed here becomes public; the sysfs and interrupts snapshots are a full environment description of a personal machine |

## STRIDE threat register

| Threat ID | Category | Component | Disposition | Mitigation plan |
|-----------|----------|-----------|-------------|-----------------|
| T-1-06 | Information disclosure | probe-sysfs-tuning.txt, probe-proc-interrupts.txt, FINDINGS.md | mitigate | The command list in task 3 reads no hostname, no MAC address, no serial number, no username and no wifi SSID. | T-1-14 | Tampering | rtla built from a cloned kernel tree | mitigate | Clone over HTTPS from `git.kernel.org` only, on the branch matching the running kernel major.minor. Do not add a third-party PPA and do not install a prebuilt rtla binary from an unofficial source. Record the clone URL and the resolved commit in rtla-build-log.txt. |
| T-1-15 | Spoofing | probe captures presented as measurements | mitigate | Every probe file carries the `probe-` prefix and lives under `crates/*/tests/fixtures/`, never under `measurements/`. Plan 01-09 records the rule in docs/publication-layout.md and plan 01-08 enforces the directory separation in CI. |
