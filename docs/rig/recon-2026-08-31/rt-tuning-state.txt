=== unit file (/etc/systemd/system/rt-tuning.service) ===
[Unit]
Description=neurorust real-time tuning (governor, turbo)
After=multi-user.target

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStart=/usr/local/sbin/rt-tuning.sh

[Install]
WantedBy=multi-user.target

=== /usr/local/sbin/rt-tuning.sh (the ExecStart script) ===
#!/usr/bin/env bash
for g in /sys/devices/system/cpu/cpu*/cpufreq/scaling_governor; do
  echo performance > "$g" 2>/dev/null || true
done
echo 1 > /sys/devices/system/cpu/intel_pstate/no_turbo 2>/dev/null || true
